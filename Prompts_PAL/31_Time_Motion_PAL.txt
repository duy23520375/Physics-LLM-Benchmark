def solution():
    import sympy as sp
    
    # Khai báo tham số đầu vào
    delta_t1 = sp.Rational(1, 12)  # s
    S = 6.0                        # cm
    
    # Bước 1: Tính chu kỳ T và tần số góc omega
    T = 6.0 * delta_t1

    # Bước 2: Tính biên độ A từ phương trình S = A * (2 - sqrt(3))
    A = S / (2 - sp.sqrt(3))
    
    # Bước 3: Tính tốc độ cực đại v_max = omega * A
    v_max = omega * A

    return result
