LỜI GIẢI:
[1. GIẢI THÍCH LÝ THUYẾT]:
- Điều kiện đề bài: Wt <= 3.Wđ. 
- Ta có cơ năng W = Wt + Wđ = Wt + Wt/3 = 4.Wt/3 => Wt/W <= 3/4.
- Theo công thức thế năng: Wt/W = x^2/A^2 => x^2/A^2 <= 3/4 => |x| <= A.sqrt(3)/2.
- Như vậy, vùng cần xét là khoảng li độ trong đoạn [-A.sqrt(3)/2, A.sqrt(3)/2].

def solution():
    import sympy as sp
    
    # Khai báo tham số đầu vào
    v_tb = 150 * sp.sqrt(3)  # cm/s
    
    # Bước 1: Trong nửa chu kỳ (T/2), thời gian vật ở vùng |x| <= A*sqrt(3)/2 là delta_t (tính theo T)
    # delta_t_factor = 2 * (1/6) = 1/3 (tức delta_t = T/3)
    delta_t_factor = sp.Rational(1, 3)
    
    # Bước 2: Quãng đường đi được từ -A*sqrt(3)/2 đến A*sqrt(3)/2 là S (tính theo A)
    # S_factor = sqrt(3) (tức S = A*sqrt(3))
    S_factor = sp.sqrt(3)
    
    # Bước 3: Tính tỉ số A_over_T từ công thức v_tb = S / delta_t = (S_factor / delta_t_factor) * (A / T)
    ratio_factor = S_factor / delta_t_factor
    A_over_T = v_tb / ratio_factor
    
    # Bước 4: Tính tốc độ cực đại v_max = 2 * pi * (A / T)
    v_max = 2 * sp.pi * A_over_T
    
    result = str(v_max)
    return result

def solution():
    import sympy as sp
    
    # Khai báo tham số đầu vào
    v_tb = 300.0  # cm/s
    
    # Bước 1: Trong nửa chu kỳ, thời gian vật ở vùng |x| <= A/2 là delta_t (tính theo T)
    # delta_t_factor = 2 * (1/12) = 1/6 (tức delta_t = T/6)
    delta_t_factor = sp.Rational(1, 6)
    
    # Bước 2: Quãng đường đi được từ -A/2 đến A/2 là S (tính theo A)
    # S_factor = 1.0 (tức S = A)
    S_factor = 1.0
    
    # Bước 3: Tính tỉ số A_over_T từ công thức v_tb = S / delta_t
    ratio_factor = S_factor / delta_t_factor
    A_over_T = v_tb / ratio_factor
    
    # Bước 4: Tính tốc độ cực đại v_max = 2 * pi * (A / T)
    v_max = 2 * sp.pi * A_over_T
    
    result = str(v_max)
    return result
