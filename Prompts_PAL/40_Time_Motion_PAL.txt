[1. GIẢI THÍCH LÝ THUYẾT]:
- Thông số ban đầu: Biên độ A = 8 cm, Tần số góc omega = 2.pi (rad/s), pha ban đầu của li độ phi_x = pi/4 rad.

def solution():
    import sympy as sp
    
    # Khai báo tham số đầu vào
    omega = 2 * sp.pi
    A = 8.0
    phi_x = sp.pi / 4           # Pha ban đầu của li độ x
    v_target = -8 * sp.pi
    v_max = omega * A
    
    # Bước 1: Tính pha ban đầu của hàm vận tốc phi_v = phi_x + pi/2
    phi_v = phi_x + sp.pi / 2
    
    # Bước 2: Tính góc alpha = arccos(v_target / v_max)
    cos_val = v_target / v_max
    alpha = sp.acos(cos_val)    # 2*pi/3
    
    # Bước 3: Chọn nghiệm pha âm (v đang tăng): omega * t + phi_v = -alpha + 2*k*pi
    # => t = (-alpha - phi_v + 2*k*pi) / omega
    k = sp.Symbol('k', integer=True)
    t_expr = (-alpha - phi_v + 2 * k * sp.pi) / omega
    
    # Bước 4: Tìm nghiệm t > 0 nhỏ nhất (k = 1)
    t_first = t_expr.subs(k, 1)
    
    result = str(sp.nsimplify(t_first))
    return result

def solution():
    import sympy as sp
    
    # Khai báo tham số đầu vào
    omega = 4 * sp.pi
    A = 10.0
    phi_x = sp.pi / 6           # Pha ban đầu của x
    v_max = omega * A
    v_target = -20 * sp.pi * sp.sqrt(3)
    
    # Bước 1: Tính pha ban đầu của hàm vận tốc
    phi_v = phi_x + sp.pi / 2
    
    # Bước 2: Tính góc alpha = arccos(v_target / v_max)
    cos_val = -sp.sqrt(3) / 2
    alpha = sp.acos(cos_val)    # 5*pi/6
    
    # Bước 3: Thiết lập biểu thức nghiệm với pha âm
    k = sp.Symbol('k', integer=True)
    t_expr = (-alpha - phi_v + 2 * k * sp.pi) / omega
    
    # Bước 4: Tìm nghiệm t > 0 nhỏ nhất (k = 1)
    t_first = t_expr.subs(k, 1)
    
    result = str(sp.nsimplify(t_first))
    return result
