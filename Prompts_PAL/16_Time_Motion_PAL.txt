[1. GIẢI THÍCH LÝ THUYẾT]:
Tốc độ trung bình (v_tb) tính theo quãng đường S, vận tốc trung bình (v_vt) tính theo độ dời delta_x. Tại t = 0, pha ban đầu bằng 0 nên vật ở biên dương x1 = A.

def solution():
    # Khai báo tỉ lệ theo A (chọn chuẩn hóa A = 1.0)
    A = 1.0
    x1 = A                      # Tại t=0 ở biên dương
    
    # Bước 1: Xác định tọa độ x2 sau 3T/4 và tính độ dời delta_x
    x2 = 0.0                    # VTCB
    delta_x = x2 - x1           # delta_x = -A
    
    # Bước 2: Tính tổng quãng đường S (3T/4 = T/2 + T/4)
    S = 2 * A + A               # S = 3A
    
    # Bước 3: Lập tỉ số giữa tốc độ trung bình và độ lớn vận tốc trung bình
    ratio = S / abs(delta_x)
    
    result = float(ratio)
    return result

def solution():
    # Khai báo tham số đầu vào

    x1 = -A                     # Tại t=0 ở biên âm (-6 cm)

    # Khai báo tham số đầu vào
    A = 4.0                     # cm
    x1 = 0.0                    # Tại t=0 ở VTCB theo chiều dương
    
    # Bước 1: Xác định tọa độ x2 sau 5T/4 (T + T/4) và tính độ dời delta_x

    delta_x = x2 - x1           # delta_x = 4 - 0 = 4 cm
    
    # Bước 2: Tính tổng quãng đường S (5T/4 = T + T/4)
    S = 4 * A + A               # S = 16 + 4 = 20 cm
    
    # Bước 3: Lập tỉ số

    return result
