def solution():
    import sympy as sp
    
    # Khai báo tham số đầu vào
    A_sq = 100.0
    v_max_sq = 4000.0
    t_val = sp.Rational(1, 2)
    
    # Bước 1: Tính biên độ A và tần số góc omega
    A = sp.sqrt(A_sq)
    omega_sq = v_max_sq / A_sq
    # omega_sq = 40 = 4 * 10 = 4 * pi^2 => omega = 2*pi
    omega = 2 * sp.pi
    
    # Bước 2: Xác định pha tại thời điểm t (phi_t = pi/4 do x/A = sqrt(2)/2 và v < 0)
    phi_t = sp.pi / 4
    
    # Bước 3: Tìm pha ban đầu phi = phi_t - omega * t
    phi = phi_t - omega * t_val
    
    result = {"A": int(A), "omega": str(omega), "phi": str(phi)}
    return result
