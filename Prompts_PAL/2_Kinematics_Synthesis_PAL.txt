def solution():
    import sympy as sp
    t = sp.Symbol('t', real=True)
    
    # Biến đổi lượng giác: -cos(alpha) = cos(alpha + pi)
    # x = -8*cos(2*pi*t) = 8*cos(2*pi*t + pi)
    # Bước 1: Trích xuất biên độ A (A > 0) và pha ban đầu phi
    A = 8.0          # cm
    phi = sp.pi      # rad
    
    return {"A": A, "phi": str(phi)}
