def solution():
    import sympy as sp
    
    # Khai báo tham số đầu vào
    omega = 2 * sp.pi       # rad/s

    # Bước 2: Tính số chu kỳ nguyên k = (N - 1) // 2
    k = (N - 1) // 2

    t1 = float(delta_phi / omega)
