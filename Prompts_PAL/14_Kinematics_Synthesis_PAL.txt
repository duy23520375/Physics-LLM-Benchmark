def solution():
    import sympy as sp
    
    # Khai báo tham số đầu vào
    A1 = 3.0                   # cm
    phi1 = sp.pi / 2           # rad
    phi2 = -sp.pi / 6          # rad
    omega = 20.0               # rad/s
    a_max = 2000 * sp.sqrt(3)  # cm/s^2
    
    # Bước 1: Tính biên độ tổng hợp A = a_max / omega^2
    A = a_max / (omega**2)
    
    # Bước 2: Tính độ lệch pha delta_phi = phi2 - phi1
    delta_phi = phi2 - phi1
    
    # Bước 3: Thiết lập và giải phương trình bậc 2 theo a: a^2 + (2*A1*cos(delta_phi))*a + (A1^2 - A^2) = 0
    a_sym = sp.Symbol('a', real=True)
    eq = sp.Eq(A**2, A1**2 + a_sym**2 + 2 * A1 * a_sym * sp.cos(delta_phi))
    roots = sp.solve(eq, a_sym)
    
    # Bước 4: Lọc nghiệm dương (a > 0)
    valid_roots = [float(r.evalf()) for r in roots if r > 0]
    a_val = valid_roots[0] if valid_roots else None
    
    result = round(a_val, 2)
    return result
