- Áp dụng hệ thức độc lập với thời gian cho hai trạng thái dao động: x1^2 + (v1 / omega)^2 = x2^2 + (v2 / omega)^2 = A^2.

def solution():
    import sympy as sp
    
    # Khai báo tham số đầu vào
    x1 = 3.0
    v1 = -40.0 * sp.sqrt(3)
    x2 = 3.0 * sp.sqrt(2)
    v2 = 40.0 * sp.sqrt(2)
    
    # Bước 1: Tính hiệu bình phương vận tốc và hiệu bình phương li độ
    delta_v2 = v1**2 - v2**2
    delta_x2 = x2**2 - x1**2
    
    # Bước 2: Tính tần số góc omega = sqrt(delta_v2 / delta_x2)
    omega = sp.sqrt(delta_v2 / delta_x2)
    
    # Bước 3: Tính biên độ A = sqrt(x1^2 + (v1/omega)^2)
    A = sp.sqrt(x1**2 + (v1 / omega)**2)
    
    result = {"A": float(A), "omega": str(omega)}
    return result
